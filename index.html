// Waffle Ball ABS (Automated Ball-Strike) System // Track balls, strikes, and outs directly from your browser. class WaffleBallABS { constructor(maxBalls = 4, maxStrikes = 3, maxOuts = 3) { this.maxBalls = maxBalls; this.maxStrikes = maxStrikes; this.maxOuts = maxOuts; this.balls = 0; this.strikes = 0; this.outs = 0; this.inning = 1; this.isTopInning = true; this.homeScore = 0; this.awayScore = 0; } // Pitch calls callPitch(result) { if (result === 'ball') { this.balls++; if (this.balls >= this.maxBalls) { console.log('WALK!'); this.resetCount(); } } else if (result === 'strike') { this.strikes++; if (this.strikes >= this.maxStrikes) { console.log('STRIKE THREE - YOU\'RE OUT!'); this.addOut(); } } else if (result === 'foul') { if (this.strikes < this.maxStrikes - 1) { this.strikes++; } } else if (result === 'hit' || result === 'out') { if (result === 'out') { this.addOut(); } else { this.resetCount(); } } this.displayStatus(); } addOut() { this.outs++; this.resetCount(); if (this.outs >= this.maxOuts) { console.log('THREE OUTS - CHANGE SIDES!'); this.outs = 0; if (!this.isTopInning) { this.inning++; } this.isTopInning = !this.isTopInning; } } addRun(team) { if (team === 'home') this.homeScore++; if (team === 'away') this.awayScore++; this.displayStatus(); } resetCount() { this.balls = 0; this.strikes = 0; } displayStatus() { const half = this.isTopInning ? 'TOP' : 'BOT'; console.log(`-----------------------------------`); console.log(`INNING: ${half} ${this.inning} | SCORE: Away ${this.awayScore} - Home ${this.homeScore}`); console.log(`COUNT: ${this.balls} Balls, ${this.strikes} Strikes | OUTS: ${this.outs}`); console.log(`-----------------------------------`); } } // Example usage: const game = new WaffleBallABS(); // Simulate pitches: game.callPitch('ball'); // Ball 1 game.callPitch('strike'); // Ball 1, Strike 1 game.callPitch('strike'); // Ball 1, Strike 2 game.callPitch('strike'); // Strikeout!
